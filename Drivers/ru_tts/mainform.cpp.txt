char* Win1251toKoi8(char *src)
{
    wchar_t *str_unicode = NULL;
    char * str_KOI;
    int size = MultiByteToWideChar(1251, 0, src, -1, str_unicode, 0);
    if (size){
        str_unicode = new wchar_t[size];
        if (MultiByteToWideChar(1251, 0, src, -1, str_unicode, size)){
            size = WideCharToMultiByte(20866, 0, str_unicode, -1, str_KOI,0, NULL, NULL);
            if (size){
                str_KOI = new char[size];
                size = WideCharToMultiByte(20866, 0, str_unicode, -1, str_KOI, size, NULL, NULL);
            }
        }
    }
    return str_KOI;
}

int transcription_out_function(void *buffer, size_t size, void *user_data)
{
    char *buf = (char *) buffer;
    AnsiString str = "";
    for (int i=0; i<size; i++){
        str = str + phonems[buf[i]];
    }
    Form1->txtTranscript->Lines->Add(str);

int populateVectors (void *buffer, size_t size, void *user_data)
{
    //будет вызываться когда мы синтезируем данные для микроконтроллера
    //ах да, нельзя нам уродовать эти данные!
    for (int i=0; i<ttscb.script.length; i++){
        ttscb.script.sounds[i].stage += intonations.size(); //т.е когда у нас пошла вторая фраза, и у нас уже сидит 12 "интонаций", мы сменим индекс с 0 на 12.
        sounds.push_back(ttscb.script.sounds[i]);
        ttscb.script.sounds[i].stage -= intonations.size(); //т.е когда у нас пошла вторая фраза, и у нас уже сидит 12 "интонаций", мы сменим индекс с 0 на 12.
    }
    for (int i=0; i<NSTAGES; i++){
        intonations.push_back(ttscb.script.icb[i]);
    }

}
//продолжаем костылять. Лениво было ещё один тип функций объявлять. Короче, когда оно вызывается, мы лезем не в буфер, а в soundscript.
int beforeMakeSound (void *buffer, size_t size, void *user_data)
{
    AnsiString tmp;
    int8_t stageUsed[12];
    memset (stageUsed, 0, 12);
    int8_t TruncatedBigChunks = 0;
    int16_t scnt;

    TotalSoundCount +=ttscb.script.length;
    Form1->txtSoundscript->Lines->Add("");
    Form1->txtSoundscript->Lines->Add("");
    Form1->txtSoundscript->Lines->Add("---- Новая фраза ----");


    for (int i=0; i<ttscb.script.length; i++){
        tmp.sprintf("ВФ %d (%s), длительность %d, интон отрезок %d", ttscb.script.sounds[i].id, WFs[ttscb.script.sounds[i].id], ttscb.script.sounds[i].duration, ttscb.script.sounds[i].stage);
        Form1->txtSoundscript->Lines->Add(tmp);
        stageUsed[ttscb.script.sounds[i].stage] = 1;
        if (ttscb.script.sounds[i].id < 169) {
            scnt = ttscb.script.voice->sound_lengths[ttscb.script.sounds[i].id];
            if ((scnt!=51)&&(scnt!=101)&&(ttscb.script.sounds[i].duration < scnt))
                TruncatedBigChunks = 1;
        }
    }
    Form1->txtSoundscript->Lines->Add("");
    if(TruncatedBigChunks != 0)
        Form1->txtSoundscript->Lines->Add("Часть согласных звуков, не участвующих в тонообразовании, была обрезана");
    Form1->txtSoundscript->Lines->Add("--- интонационные отрезки ---");
    for (int i=0; i<12; i++){
        if (stageUsed[i]) {
            tmp.sprintf("Интон. отрезок %d: исх тон %d (в сэмплах), дельта %d, период %d", i, ttscb.script.icb[i].stretch, ttscb.script.icb[i].delta, ttscb.script.icb[i].period);
            Form1->txtSoundscript->Lines->Add(tmp);
        }
    }

    //а теперь будем наполнять вектор
    for (int i=0; i<ttscb.script.length; i++){
        sounds.push_back(ttscb.script.sounds[i]);
    }



    return 0;
}

void analyzeUsedSamples()
{
    int curCnt;
    for (int i = 0; i < 169; i++){  //идём по ВФ
        curCnt = 0;
        for (int j = ttscb.script.voice->sound_offsets[i]; j < ttscb.script.voice->sound_offsets[i+1]; j++) {
            if (used_samples[j])
                curCnt++;
        }
//        Form1->txtTranscript->Lines->Add("ВФ "+IntToStr(i)+": "+IntToStr(curCnt)+" сэмплов");
        Form1->txtTranscript->Lines->Add(IntToStr(i)+"\t"+IntToStr(curCnt));
    }
}
